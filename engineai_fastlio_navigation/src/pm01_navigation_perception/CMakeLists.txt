cmake_minimum_required(VERSION 3.8)
project(pm01_navigation_perception)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclpy REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
add_executable(custommsg_to_cloud src/cloud/custommsg_to_cloud.cpp)
ament_target_dependencies(custommsg_to_cloud rclcpp sensor_msgs livox_ros_driver2)
add_executable(lio_cloud_adapter src/cloud/lio_cloud_adapter.cpp)
ament_target_dependencies(lio_cloud_adapter rclcpp sensor_msgs tf2)
install(TARGETS custommsg_to_cloud lio_cloud_adapter DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS FILES_MATCHING PATTERN "*.py")
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_scan_projection test/test_scan_projection.py)
endif()
ament_package()
