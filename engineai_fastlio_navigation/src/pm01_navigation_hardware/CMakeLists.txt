cmake_minimum_required(VERSION 3.8)
project(pm01_navigation_hardware)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_package(rclcpp REQUIRED)
find_package(action_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(interface_protocol REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(cmd_vel_bridge src/motion/cmd_vel_bridge.cpp)
ament_target_dependencies(cmd_vel_bridge rclcpp action_msgs geometry_msgs interface_protocol std_srvs)
target_include_directories(cmd_vel_bridge PRIVATE include)
add_executable(odom_relay src/odometry/odom_relay.cpp)
ament_target_dependencies(odom_relay rclcpp nav_msgs tf2 tf2_ros tf2_geometry_msgs)
target_link_libraries(odom_relay Eigen3::Eigen)
install(TARGETS cmd_vel_bridge odom_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(body_velocity_command test/test_body_velocity_command.cpp)
  target_include_directories(body_velocity_command PRIVATE include)
endif()
ament_package()
