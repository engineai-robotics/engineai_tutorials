cmake_minimum_required(VERSION 3.8)
project(pm01_navigation_sim)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclpy REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(sim_cmd_vel_adapter src/motion/sim_cmd_vel_adapter.cpp)
ament_target_dependencies(sim_cmd_vel_adapter rclcpp geometry_msgs diagnostic_msgs nav_msgs std_srvs)
add_executable(sim_body_tf src/odometry/sim_body_tf.cpp)
ament_target_dependencies(sim_body_tf rclcpp nav_msgs tf2 tf2_ros tf2_geometry_msgs)
install(TARGETS sim_cmd_vel_adapter sim_body_tf DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS FILES_MATCHING PATTERN "*.py")
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_interfaces test/test_interfaces.py)
endif()
ament_package()
