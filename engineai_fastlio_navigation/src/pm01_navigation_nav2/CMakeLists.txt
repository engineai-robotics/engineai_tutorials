cmake_minimum_required(VERSION 3.8)
project(pm01_navigation_nav2)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
foreach(dep rclcpp dwb_core pluginlib nav2_controller nav2_core nav2_rotation_shim_controller tf2 tf2_ros tf2_geometry_msgs diagnostic_msgs geometry_msgs behaviortree_cpp_v3)
  find_package(${dep} REQUIRED)
endforeach()
add_library(pm01_dwb_critics SHARED src/critics/walking_command_critic.cpp)
target_include_directories(pm01_dwb_critics PRIVATE include)
ament_target_dependencies(pm01_dwb_critics rclcpp dwb_core pluginlib)
pluginlib_export_plugin_description_file(dwb_core critic_plugins.xml)
add_library(pm01_nav_controllers SHARED src/controllers/settling_dwb_controller.cpp src/controllers/latest_tf_rotation_shim.cpp)
target_include_directories(pm01_nav_controllers PRIVATE include)
ament_target_dependencies(pm01_nav_controllers rclcpp dwb_core nav2_controller nav2_rotation_shim_controller pluginlib tf2_geometry_msgs)
add_library(pm01_goal_checkers SHARED src/goal_checkers/settled_goal_checker.cpp)
target_include_directories(pm01_goal_checkers PRIVATE include)
ament_target_dependencies(pm01_goal_checkers rclcpp nav2_controller nav2_core pluginlib)
pluginlib_export_plugin_description_file(nav2_core controller_plugins.xml)
install(TARGETS pm01_dwb_critics pm01_nav_controllers pm01_goal_checkers LIBRARY DESTINATION lib)
add_executable(navigation_command_gate src/navigation_command_gate.cpp src/navigation_command_gate_main.cpp)
target_include_directories(navigation_command_gate PRIVATE include)
ament_target_dependencies(navigation_command_gate rclcpp diagnostic_msgs geometry_msgs)
add_library(pm01_navigation_ready_wait_bt_node SHARED src/navigation_ready_wait.cpp)
target_include_directories(pm01_navigation_ready_wait_bt_node PRIVATE include)
target_compile_definitions(pm01_navigation_ready_wait_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(pm01_navigation_ready_wait_bt_node rclcpp diagnostic_msgs behaviortree_cpp_v3)
install(TARGETS navigation_command_gate DESTINATION lib/${PROJECT_NAME})
install(TARGETS pm01_navigation_ready_wait_bt_node LIBRARY DESTINATION lib)
install(DIRECTORY behavior_trees DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  foreach(test quiet_window walking_command_limits navigation_readiness)
    ament_add_gtest(${test} test/test_${test}.cpp)
    target_include_directories(${test} PRIVATE include)
  endforeach()
  ament_add_gtest(goal_completion test/test_goal_completion.cpp)
  ament_target_dependencies(goal_completion rclcpp nav2_core pluginlib tf2_geometry_msgs)
  ament_add_gtest(plugin_loading test/test_plugin_loading.cpp)
  ament_target_dependencies(plugin_loading nav2_core dwb_core pluginlib)
  ament_add_gtest(navigation_ready_wait test/test_navigation_ready_wait.cpp src/navigation_ready_wait.cpp)
  target_include_directories(navigation_ready_wait PRIVATE include)
  ament_target_dependencies(navigation_ready_wait rclcpp diagnostic_msgs behaviortree_cpp_v3)
  target_compile_definitions(navigation_ready_wait PRIVATE
    PM01_READY_BT_LIBRARY="$<TARGET_FILE:pm01_navigation_ready_wait_bt_node>")
  add_dependencies(navigation_ready_wait pm01_navigation_ready_wait_bt_node)
  ament_add_gtest(navigation_command_gate_test test/test_navigation_command_gate.cpp src/navigation_command_gate.cpp)
  target_include_directories(navigation_command_gate_test PRIVATE include)
  ament_target_dependencies(navigation_command_gate_test rclcpp diagnostic_msgs geometry_msgs)
endif()
ament_package()
